project(NexvaryDiskCare VERSION 0.8.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)

qt_standard_project_setup(REQUIRES 6.8)
add_library(dc_core STATIC src/core/Operations.cpp src/core/StoragePolicy.cpp src/core/CrystalReport.cpp src/core/ReadCoverage.cpp src/core/MediaRead.cpp src/core/Recovery.cpp src/core/FatRecovery.cpp src/core/Partitions.cpp src/core/FilesystemRecovery.cpp src/core/RescueEngine.cpp)

target_compile_definitions(nexvary_dc PRIVATE DC_APP_VERSION="${PROJECT_VERSION}")
target_link_libraries(nexvary_dc PRIVATE dc_core Qt6::Concurrent Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Sql)
qt_add_qml_module(nexvary_dc URI Nexvary.DC VERSION 1.0 QML_FILES qml/Main.qml qml/ActionButton.qml qml/Panel.qml qml/DiskIcon.qml qml/AppIcon.qml qml/AboutWorkspace.qml qml/StorageWorkspace.qml qml/BootRepairWorkspace.qml qml/SmartWorkspace.qml qml/WindowsRecoveryWorkspace.qml qml/ResultPanel.qml)
file(GLOB DC_ICON_FILES CONFIGURE_DEPENDS assets/icons/*.png)
qt_add_resources(nexvary_dc cdi PREFIX "/engines" BASE packaging FILES packaging/cdi-files.json)
qt_add_resources(nexvary_dc icons PREFIX "/assets" BASE assets FILES ${DC_ICON_FILES})
qt_add_resources(nexvary_dc storage PREFIX "/storage" BASE scripts FILES scripts/storage.ps1 scripts/policy.ps1 scripts/hybrid.ps1 scripts/boot_repair.py scripts/firmware_boot.ps1 scripts/windows_recovery.ps1)
if(WIN32)
 target_link_libraries(nexvary_dc PRIVATE shell32 advapi32)
 configure_file(packaging/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
 target_sources(nexvary_dc PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
 set_target_properties(nexvary_dc PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
