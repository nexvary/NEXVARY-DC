cmake_minimum_required(VERSION 3.24)
project(NexvaryDiskCare VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Concurrent Gui Qml Quick QuickControls2 Sql Test)
qt_standard_project_setup(REQUIRES 6.8)
add_library(dc_core STATIC src/core/Operations.cpp src/core/StoragePolicy.cpp src/core/MediaRead.cpp src/core/Recovery.cpp src/core/FatRecovery.cpp)
target_include_directories(dc_core PUBLIC src/core)
target_link_libraries(dc_core PUBLIC Qt6::Core)
qt_add_executable(nexvary_dc src/app/main.cpp src/app/Controller.cpp src/app/Controller.h)
target_link_libraries(nexvary_dc PRIVATE dc_core Qt6::Concurrent Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Sql)
qt_add_qml_module(nexvary_dc URI Nexvary.DC VERSION 1.0 QML_FILES qml/Main.qml qml/ActionButton.qml qml/Panel.qml qml/DiskIcon.qml qml/AppIcon.qml qml/AboutWorkspace.qml qml/StorageWorkspace.qml qml/ResultPanel.qml)
file(GLOB DC_ICON_FILES CONFIGURE_DEPENDS assets/icons/*.png)
qt_add_resources(nexvary_dc icons PREFIX "/assets" BASE assets FILES ${DC_ICON_FILES})
qt_add_resources(nexvary_dc storage PREFIX "/storage" BASE scripts FILES scripts/storage.ps1 scripts/policy.ps1 scripts/hybrid.ps1)
if(WIN32)
 target_link_libraries(nexvary_dc PRIVATE shell32 advapi32)
 configure_file(packaging/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
 target_sources(nexvary_dc PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
 set_target_properties(nexvary_dc PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
include(CTest)
if(BUILD_TESTING)
 qt_add_executable(dc_policy_tests tests/test_policy.cpp)
 target_link_libraries(dc_policy_tests PRIVATE dc_core Qt6::Test)
 add_test(NAME storage_policy COMMAND dc_policy_tests)
 if(WIN32)
  add_test(NAME powershell_policy COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_storage_policy.ps1)
 endif()
 qt_add_executable(dc_recovery_tests tests/test_recovery.cpp)
 target_link_libraries(dc_recovery_tests PRIVATE dc_core Qt6::Test)
 add_test(NAME recovery COMMAND dc_recovery_tests)
 if(WIN32)
  add_test(NAME boot_helpers COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_boot.ps1)
 endif()
 qt_add_executable(dc_tests tests/test_core.cpp)
 target_link_libraries(dc_tests PRIVATE dc_core Qt6::Test)
 add_test(NAME core COMMAND dc_tests)
 add_test(NAME ui_startup COMMAND nexvary_dc --smoke-test)
 add_test(NAME ui_capacity_ar COMMAND nexvary_dc --smoke-test --page 5 --compact)
 add_test(NAME ui_confirmation COMMAND nexvary_dc --smoke-test --confirmation-proof --compact)
 add_test(NAME ui_storage COMMAND nexvary_dc --smoke-test --page 7 --compact)
 add_test(NAME ui_boot COMMAND nexvary_dc --smoke-test --page 8 --english --compact)
 add_test(NAME ui_about COMMAND nexvary_dc --smoke-test --page 9 --compact)
 add_test(NAME ui_capacity_en COMMAND nexvary_dc --smoke-test --page 5 --english --compact)
 qt_add_executable(dc_controller_tests tests/test_controller.cpp src/app/Controller.cpp src/app/Controller.h)
 target_include_directories(dc_controller_tests PRIVATE src/app)
 target_link_libraries(dc_controller_tests PRIVATE dc_core Qt6::Concurrent Qt6::Sql Qt6::Test)
 qt_add_resources(dc_controller_tests storage_test PREFIX "/storage" BASE scripts FILES scripts/storage.ps1 scripts/policy.ps1 scripts/hybrid.ps1)
 if(WIN32)
  target_link_libraries(dc_controller_tests PRIVATE shell32 advapi32)
 endif()
 add_test(NAME controller COMMAND dc_controller_tests)
 set_tests_properties(core controller storage_policy PROPERTIES TIMEOUT 45)
 set_tests_properties(ui_startup ui_capacity_ar ui_capacity_en ui_storage ui_boot ui_about ui_confirmation PROPERTIES TIMEOUT 45 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
install(TARGETS nexvary_dc RUNTIME DESTINATION bin)
